package main

import (
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"os"

	"github.com/JuanHuaXu/eventframed/internal/epistemic"
	"github.com/JuanHuaXu/eventframed/internal/model"
)

type fact struct {
	ID      string `json:"id"`
	Mission string `json:"mission"`
	Event   string `json:"event"`
	Target  string `json:"target"`
	Date    string `json:"date"`
	Source  string `json:"source"`
}
type result struct {
	Fact       string `json:"fact"`
	Mode       string `json:"mode"`
	SameGroup  bool   `json:"same_group"`
	Correlated bool   `json:"correlated"`
}

func check(err error) {
	if err != nil {
		panic(err)
	}
}
func main() {
	if len(os.Args) != 2 {
		panic("expected public facts path")
	}
	input, err := os.ReadFile(os.Args[1])
	check(err)
	var facts []fact
	check(json.Unmarshal(input, &facts))
	field := func(s string) model.Field { return model.Field{Value: s, Source: model.SourceObserved, Confidence: 1} }
	var results []result
	for _, f := range facts {
		origin := sha256.Sum256([]byte(f.Source))
		base := model.Event{ID: f.ID, TenantID: "public-research", SessionID: "fetch-a", Kind: "external_observation",
			Who: field(f.Mission), What: field(f.Event), Where: field(f.Target), When: field(f.Date), How: field("archive retrieval"),
			Provenance: model.Provenance{Producer: "public-fixture-adapter", ToolCallID: "fetch-a", RunID: "run-a", SourceEventIDs: []string{"public-origin-" + hex.EncodeToString(origin[:])}}}
		for _, mode := range []string{"stable_origin", "missing_origin", "producer_change", "per_fetch_origin"} {
			left, right := base, base
			right.ID = f.ID + "-repeat"
			right.SessionID = "fetch-b"
			right.Provenance.RunID = "run-b"
			right.Provenance.ToolCallID = "fetch-b"
			switch mode {
			case "missing_origin":
				left.Provenance.SourceEventIDs = nil
				right.Provenance.SourceEventIDs = nil
			case "producer_change":
				right.Provenance.Producer = "second-fixture-adapter"
			case "per_fetch_origin":
				right.Provenance.SourceEventIDs = []string{"different-fetch-origin"}
			}
			a, b := epistemic.Describe(left), epistemic.Describe(right)
			results = append(results, result{f.ID, mode, a.EvidenceGroupKey == b.EvidenceGroupKey, epistemic.Correlated(a, b, .8)})
		}
	}
	hashes := map[string]string{}
	for _, path := range []string{os.Args[1], "internal/epistemic/evidence.go", "research/public-provenance/main.go", "research/public-task-pilot/PUBLIC_PROVENANCE_PROTOCOL.md"} {
		b, e := os.ReadFile(path)
		check(e)
		h := sha256.Sum256(b)
		hashes[path] = hex.EncodeToString(h[:])
	}
	out := struct {
		Scope   string            `json:"scope"`
		Hashes  map[string]string `json:"hashes"`
		Results []result          `json:"results"`
	}{
		"Direct evidence-grouping boundary on repeated public facts; not authentication, extraction, or end-to-end confidence validation", hashes, results}
	b, err := json.MarshalIndent(out, "", "  ")
	check(err)
	fmt.Println(string(b))
}
