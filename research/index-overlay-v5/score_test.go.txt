package libravdb

import (
    "context"
    "fmt"
    "testing"
    "github.com/xDarkicex/libravdb/internal/util"
)

func TestResearchOverlayCanonicalPublicScore(t *testing.T) {
    ctx := context.Background()
    db, err := Open(WithStoragePath(t.TempDir()+"/scores.libravdb"))
    if err != nil { t.Fatal(err) }
    defer db.Close()
    col, err := db.CreateCollection(ctx, "events", WithDimension(128), WithHNSW(16,100,100), WithScalarQuantization(8,.05))
    if err != nil { t.Fatal(err) }
    if err := db.WithTx(ctx,func(tx Tx)error {
        for i:=0;i<80;i++ { if err:=tx.Insert(ctx,"events",fmt.Sprintf("e-%04d",i),overlayDBVector(i,128),nil);err!=nil{return err} }
        return nil
    });err!=nil { t.Fatal(err) }
    if err := db.WithTx(ctx,func(tx Tx)error { return tx.Insert(ctx,"events","overlay",overlayDBVector(80,128),nil) });err!=nil { t.Fatal(err) }
    metric,err := util.GetDistanceFunc(util.DistanceMetric(col.config.Metric))
    if err!=nil { t.Fatal(err) }
    mismatches := 0
    for j:=0;j<4;j++ {
        query := overlayDBVector(10000+j,128)
        result,err := col.Search(ctx,query,81)
        if err!=nil { t.Fatal(err) }
        if len(result.Results)!=81 { t.Fatal("lost candidate",len(result.Results)) }
        for _,r:=range result.Results {
            canonical,err := col.Get(ctx,r.ID)
            if err!=nil { t.Fatal(err) }
            distance := metric(vectorForIndex(col.config.Metric,query),vectorForIndex(col.config.Metric,canonical.Vector))
            want := publicScore(col.config.Metric,distance)
            if r.Score!=want { mismatches++; if mismatches==1 { t.Logf("canonical mismatch: got=%.9g want=%.9g",r.Score,want) } }
        }
    }
    if mismatches>0 { t.Fatalf("%d scores do not use canonical public metric",mismatches) }
}
