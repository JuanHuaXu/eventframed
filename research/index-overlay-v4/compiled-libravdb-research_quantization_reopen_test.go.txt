package libravdb

import (
	"context"
	"fmt"
	"reflect"
	"testing"

	"github.com/xDarkicex/libravdb/internal/quant"
)

func TestResearchQuantizationDeclarationReopenV1(t *testing.T) {
	for _, sharded := range []bool{false, true} {
		for _, mode := range []string{"none", "sq8", "fsq"} {
			t.Run(fmt.Sprintf("%s/sharded=%t", mode, sharded), func(t *testing.T) {
				ctx := context.Background()
				path := t.TempDir() + "/quantization.libravdb"
				db, err := Open(WithStoragePath(path))
				if err != nil {
					t.Fatal(err)
				}
				defer func() {
					if db != nil {
						db.Close()
					}
				}()
				opts := []CollectionOption{WithHNSW(8, 40, 20), WithSharding(sharded)}
				if mode == "sq8" {
					opts = append(opts, WithScalarQuantization(8, 0.10))
				}
				if mode == "fsq" {
					opts = append(opts, WithFSQQuantization(6, 0.10, 8, 8, 8, 6))
				}
				col, err := db.EnsureCollection(ctx, "event", 4, opts...)
				if err != nil {
					t.Fatal(err)
				}
				want := col.Config().Quantization
				for i := 0; i < 16; i++ {
					id := fmt.Sprintf("row-%02d", i)
					if err := col.Insert(ctx, id, []float32{1, float32(i) / 16, 0.1, 0.2}, map[string]interface{}{"source": "public probe"}); err != nil {
						t.Fatal(err)
					}
				}
				if err := db.Close(); err != nil {
					t.Fatal(err)
				}
				db = nil
				db, err = Open(WithStoragePath(path))
				if err != nil {
					t.Fatal(err)
				}
				col, err = db.GetCollection("event")
				if err != nil {
					t.Fatal(err)
				}
				if got := col.Config(); got.Sharded != sharded || !reflect.DeepEqual(got.Quantization, want) || col.Stats(ctx).HasQuantization != (want != nil) {
					t.Fatalf("quantization declaration lost: got=%+v want=%+v sharded=%t", got.Quantization, want, got.Sharded)
				}
				for i := 0; i < 16; i++ {
					record, err := col.Get(ctx, fmt.Sprintf("row-%02d", i))
					if err != nil || record.Metadata["source"] != "public probe" {
						t.Fatal("persisted payload", i, err)
					}
				}
				if want != nil && want.Type == quant.FiniteScalarQuantization {
					copy := col.Config()
					copy.Quantization.Levels[0] = 2
					if col.Config().Quantization.Levels[0] != 8 {
						t.Fatal("config aliases persistent declaration")
					}
				}
				if err := db.Close(); err != nil {
					t.Fatal(err)
				}
				db = nil
				db, err = Open(WithStoragePath(path))
				if err != nil {
					t.Fatal(err)
				}
				col, err = db.GetCollection("event")
				if err != nil || !reflect.DeepEqual(col.Config().Quantization, want) {
					t.Fatal("second reopen lost declaration", err)
				}
			})
		}
	}
}
